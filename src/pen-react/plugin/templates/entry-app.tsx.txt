/// <reference types="vite/client" />

import type { ComponentMap, RouteComponent, RouteModule } from '@idlesummer/pen/internal'
import { render } from 'ink'
import {
  App,
  createRoleMap,
  createRouter,
  DefaultFallback,
  ErrorFallback,
  resolveDefaultExports,
} from '@idlesummer/pen/internal'

// ── Main entry point ─────────────────────────────────────────────────────

// Maps paths to module objects
const modules = import.meta.glob<RouteModule>('/__PEN_APP_DIR__/**/{page,layout,loading,error,default}.tsx', { eager: true })
const entries = Object.entries(modules).map(([path, mod]) => [path.replace('/__PEN_APP_DIR__/', ''), mod] as const)
const routeComponents = resolveDefaultExports<RouteComponent>(entries, DefaultFallback, ErrorFallback)

// `createRouter` narrows modulePaths down further to valid routes
// Modules should be already validated by buildApp
const { matcher, modulePaths } = createRouter(Object.keys(routeComponents))
const componentMap = createRoleMap(modulePaths, routeComponents) as ComponentMap

const { waitUntilExit } = render(<App matcher={matcher} componentMap={componentMap} />)
await waitUntilExit()
