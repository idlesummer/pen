/// <reference types="vite/client" />

import type { ComponentMap, RouteComponent, RouteModule } from '@idlesummer/pen/internal'
import { render } from 'ink'
import {
  App,
  createDefaultExportMap,
  createRoleMap,
  createRouter,
  DefaultFallback,
  ErrorFallback,
} from '@idlesummer/pen/internal'

// ── Main entry point ─────────────────────────────────────────────────────

// Maps paths to module objects
const modules = import.meta.glob<RouteModule>('**/{page,layout,loading,error,default}.tsx', { eager: true, base: '__PEN_APP_DIR__' })
const routeComponents = createDefaultExportMap<RouteComponent>(Object.entries(modules), DefaultFallback, ErrorFallback)

// `createRouter` narrows modulePaths down further to valid routes
// Modules should be already validated by buildApp
const filePaths = Object.keys(routeComponents)
const { matcher, modulePaths } = createRouter(filePaths)
const componentMap = createRoleMap(modulePaths, routeComponents) as ComponentMap

const { waitUntilExit } = render(<App matcher={matcher} componentMap={componentMap} />)
await waitUntilExit()
