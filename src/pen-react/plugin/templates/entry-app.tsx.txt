/// <reference types="vite/client" />

import type { ComponentMap, DefaultComponent, ErrorComponent, Fallbacks, LayoutComponent, LoadingComponent, Module, PageComponent } from '@idlesummer/pen/internal'
import { render } from 'ink'
import {
  App,
  createDefaultExportMap,
  createRouter,
  DefaultFallback,
  ErrorFallback,
  GLOBAL_DEFAULT,
  GLOBAL_ERROR,
} from '@idlesummer/pen/internal'

// ── Main entry point ─────────────────────────────────────────────────────

// One glob per role - each module lands in its own, correctly-typed bucket
// by construction, so no runtime role classification or cast is needed.
const pageModules = import.meta.glob<Module<PageComponent>>('**/page.tsx', { eager: true, base: '__PEN_APP_DIR__' })
const layoutModules = import.meta.glob<Module<LayoutComponent>>('**/layout.tsx', { eager: true, base: '__PEN_APP_DIR__' })
const loadingModules = import.meta.glob<Module<LoadingComponent>>('**/loading.tsx', { eager: true, base: '__PEN_APP_DIR__' })
const errorModules = import.meta.glob<Module<ErrorComponent>>('**/error.tsx', { eager: true, base: '__PEN_APP_DIR__' })
const defaultModules = import.meta.glob<Module<DefaultComponent>>('**/default.tsx', { eager: true, base: '__PEN_APP_DIR__' })

function toComponents<T>(modules: Record<string, Module<T>>, fallbacks?: Fallbacks<T>) {
  return createDefaultExportMap(Object.entries(modules), fallbacks)
}

type RoleModules = {
  page: Record<string, Module<PageComponent>>
  layout: Record<string, Module<LayoutComponent>>
  loading: Record<string, Module<LoadingComponent>>
  error: Record<string, Module<ErrorComponent>>
  default: Record<string, Module<DefaultComponent>>
}

/** Resolves each role's glob'd modules into its own component map, then
 *  narrows every bucket down to the paths createRouter resolved as part
 *  of a real route. Returns the matcher alongside it since it's a
 *  byproduct of that same resolution. Modules should be already
 *  validated by buildApp. */
function createApp(modules: RoleModules) {
  const pageComponents = toComponents(modules.page)
  const layoutComponents = toComponents(modules.layout)
  const loadingComponents = toComponents(modules.loading)
  const errorComponents = toComponents(modules.error, { [GLOBAL_ERROR]: ErrorFallback })
  const defaultComponents = toComponents(modules.default, { [GLOBAL_DEFAULT]: DefaultFallback })

  const filePaths = [
    ...Object.keys(pageComponents),
    ...Object.keys(layoutComponents),
    ...Object.keys(loadingComponents),
    ...Object.keys(errorComponents),
    ...Object.keys(defaultComponents),
  ]
  const { matcher, modulePaths } = createRouter(filePaths)

  const validPaths = new Set(modulePaths)
  function pickValid<T>(components: Record<string, T>): Record<string, T> {
    return Object.fromEntries(Object.entries(components).filter(([path]) => validPaths.has(path)))
  }

  const componentMap: ComponentMap = {
    page: pickValid(pageComponents),
    layout: pickValid(layoutComponents),
    loading: pickValid(loadingComponents),
    error: pickValid(errorComponents),
    default: pickValid(defaultComponents),
  }
  return { matcher, componentMap }
}

const { matcher, componentMap } = createApp({
  page: pageModules,
  layout: layoutModules,
  loading: loadingModules,
  error: errorModules,
  default: defaultModules,
})

const { waitUntilExit } = render(<App matcher={matcher} componentMap={componentMap} />)
await waitUntilExit()
