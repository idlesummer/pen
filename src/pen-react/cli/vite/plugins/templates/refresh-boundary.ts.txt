// `configureServer` initializes these before transformed modules run. Named
// `refresh`, not `RefreshRuntime`, so it can't collide with the ambient
// global of that name declared in refresh-boundary.d.ts - this file has no
// static import/export, so some tools treat its top-level declarations as
// global script scope rather than module scope.
const refresh = globalThis.RefreshRuntime!
const __prev_Reg__ = globalThis.$RefreshReg$
const __prev_Sig__ = globalThis.$RefreshSig$
globalThis.$RefreshReg$ = refresh.getRefreshReg("__PEN_REFRESH_FILENAME__")
globalThis.$RefreshSig$ = refresh.createSignatureFunctionForTransform

// __PEN_REFRESH_CODE__

globalThis.$RefreshReg$ = __prev_Reg__
globalThis.$RefreshSig$ = __prev_Sig__
if (import.meta.hot) {
  import(/* @vite-ignore */ import.meta.url).then((prevExports) => {
    import.meta.hot!.accept((nextExports) => {
      if (!nextExports) return
      const invalidateMessage = refresh.validateRefreshBoundaryAndEnqueueUpdate(
        prevExports,
        nextExports,
      )
      if (invalidateMessage)
        import.meta.hot!.invalidate(invalidateMessage)
    })
  })
}
