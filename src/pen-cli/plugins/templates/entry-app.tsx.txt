// #region strip
/// <reference types="vite/client" />
declare const __PEN_INK_OPTIONS__: import('ink').RenderOptions
// #endregion

import type {
  Module,
  PageComponent,
  LayoutComponent,
  LoadingComponent,
  ErrorComponent,
  DefaultComponent,
} from '@idlesummer/pen/internal'
import { render } from 'ink'
import { PenApp, createPenAppProps } from '@idlesummer/pen/internal'

// Vite's `import.meta.glob` needs literal arguments (anything else silently matches nothing),
// so there's one typed glob per role. The leading './' keeps `base` scoping active.
const page = import.meta.glob<Module<PageComponent>>('./**/page.tsx', {
  base: './__PEN_APP_DIR__',
  eager: true,
})
const layout = import.meta.glob<Module<LayoutComponent>>('./**/layout.tsx', {
  base: './__PEN_APP_DIR__',
  eager: true,
})
const loading = import.meta.glob<Module<LoadingComponent>>('./**/loading.tsx', {
  base: './__PEN_APP_DIR__',
  eager: true,
})
const error = import.meta.glob<Module<ErrorComponent>>('./**/error.tsx', {
  base: './__PEN_APP_DIR__',
  eager: true,
})
const default_ = import.meta.glob<Module<DefaultComponent>>('./**/default.tsx', {
  base: './__PEN_APP_DIR__',
  eager: true,
})

// ── Main entry point ─────────────────────────────────────────────────────

const penAppProps = createPenAppProps({ page, layout, loading, error, default: default_ })
const { waitUntilExit } = render(<PenApp {...penAppProps} />, __PEN_INK_OPTIONS__)
await waitUntilExit()
