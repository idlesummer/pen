// #region codegen
/// <reference types="vite/client" />
declare const __PEN_INK_OPTIONS__: import('ink').RenderOptions
declare const console: { error: (...args: unknown[]) => void }
// #endregion

import { render } from 'ink'
import { PenApp, createComponentMatcher, ROUTES_CHANGED_EVENT } from '@idlesummer/pen/internal'

// Vite's `import.meta.glob` needs literal arguments (anything else silently matches nothing),
// so there's one typed glob per role. The leading './' keeps `base` scoping active.
const moduleRegistry = {
  page:     import.meta.glob('./**/page.tsx',    { base: './__PEN_APP_DIR__', eager: true }),
  layout:   import.meta.glob('./**/layout.tsx',  { base: './__PEN_APP_DIR__', eager: true }),
  loading:  import.meta.glob('./**/loading.tsx', { base: './__PEN_APP_DIR__', eager: true }),
  error:    import.meta.glob('./**/error.tsx',   { base: './__PEN_APP_DIR__', eager: true }),
  default:  import.meta.glob('./**/default.tsx', { base: './__PEN_APP_DIR__', eager: true }),
}

// ── Main entry point ─────────────────────────────────────────────────────

// Deleting any route file Vite statically imported above (via the glob)
// makes Vite fall back to its default HMR behavior and re-run this whole
// module - there's no "accept" for a deleted dependency. Ink only supports
// one render() per process, and the live reload below already covers
// deletions on its own, so a re-run has nothing left to do - skip it,
// using a global since each re-run gets a fresh module scope of its own.
const global = globalThis as { __PEN_RENDERED__?: boolean }
if (!global.__PEN_RENDERED__) {
  global.__PEN_RENDERED__ = true

  const componentMatcher = createComponentMatcher(moduleRegistry)
  const { rerender, waitUntilExit } = render(<PenApp componentMatcher={componentMatcher} />, __PEN_INK_OPTIONS__)

  // Dev-only: the glob above only ever runs once, so a route file added or
  // removed mid-session needs a fresh load - the dev server sends just the
  // matching paths after noticing the change (not loaded modules: loading
  // them anywhere but here would produce distinct instances of shared
  // dependencies, like pen's navigation context, breaking identity-sensitive
  // state the rest of this already-running app depends on).
  type RoutePaths = Record<keyof typeof moduleRegistry, string[]>

  const loadModules = async (paths: string[]) => {
    const entries = paths.map(async path => [path, await import(/* @vite-ignore */ `/__PEN_APP_DIR__/${path}`)] as const)
    return Object.fromEntries(await Promise.all(entries))
  }

  import.meta.hot?.on(ROUTES_CHANGED_EVENT, async (paths: RoutePaths) => {
    try {
      const [page, layout, loading, error, default_] = await Promise.all([
        loadModules(paths.page),
        loadModules(paths.layout),
        loadModules(paths.loading),
        loadModules(paths.error),
        loadModules(paths.default),
      ])
      const registry = { page, layout, loading, error, default: default_ }
      rerender(<PenApp componentMatcher={createComponentMatcher(registry)} />)
    }
    catch (err) {
      // A failed reload must not take the current render down with it -
      // better stale than blank. Known open issue: deleting a route file
      // currently fails here (not yet root-caused).
      console.error(err)
    }
  })

  await waitUntilExit()
}
