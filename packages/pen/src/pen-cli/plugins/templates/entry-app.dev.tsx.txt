/// <reference types="vite/client" />

import { render } from 'ink'
import { appDir, inkOptions, initialPaths } from 'virtual:pen/runtime-config'
import { PenDevApp, createRouteRendererStore, ROUTES_CHANGED_EVENT, createRouteRenderer, createModuleRegistry, loadModules } from '@idlesummer/pen/internal'

type RoutePaths = typeof initialPaths

// No static glob here, unlike entry-app.build.tsx.txt - its mere presence
// in a module Vite transforms is enough for Vite to track these files as
// a dependency, whether or not the result is ever used. Deleting one
// would then make Vite fully re-run this whole module on its own, which
// breaks the hot listener registered below for the rest of the session
// (new events stop reaching it afterward). Loading paths dynamically, by
// request, sidesteps that tracking entirely.
const moduleRegistry = await createModuleRegistry(initialPaths, paths => loadModules(appDir, paths))
const rendererStore = createRouteRendererStore(moduleRegistry)
const { waitUntilExit } = render(<PenDevApp rendererStore={rendererStore} />, inkOptions)

// Pushing each reload into rendererStore, instead of calling Ink's
// render() again, lets React's own update scheduling handle it - the
// same way navigation actions already update the app without ever
// touching render().
import.meta.hot?.on(ROUTES_CHANGED_EVENT, async (paths: RoutePaths) => {
  try {
    const newModuleRegistry = await createModuleRegistry(paths, p => loadModules(appDir, p))
    rendererStore.setState(() => createRouteRenderer(newModuleRegistry))
  }
  catch (err) {
    console.error(err)  // a failed reload shouldn't take the current render down with it
  }
})

await waitUntilExit()
