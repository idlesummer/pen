// #region codegen
/// <reference types="@types/node" />
/// <reference types="vite/client" />
declare const __PEN_APP_DIR__: string
declare const __PEN_INK_OPTIONS__: import('ink').RenderOptions
// #endregion

import { pathToFileURL } from 'node:url'
import { render } from 'ink'
import {
  PenDevApp,
  createRouteRendererStore,
  ROUTES_CHANGED_EVENT,
  createRouteRenderer,
  createDiagnosticReport,
  _createModuleRegistry,
  type ModuleLoader,
} from '@idlesummer/pen/internal'


async function createModuleRegistryAndValidate(appDir: string) {
  const moduleLoader: ModuleLoader = path => import(/* @vite-ignore */ pathToFileURL(path).href)
  const moduleRegistry = await _createModuleRegistry(appDir, moduleLoader)
  const [warnings, error] = createDiagnosticReport(moduleRegistry, appDir)

  for (const message of warnings)
    console.warn(message)
  if (error)
    console.error(error)
  else
    return moduleRegistry
}

// ─────────────────────────────────────────────────────────────────────────
// Main entry point
// ─────────────────────────────────────────────────────────────────────────

const moduleRegistry = await createModuleRegistryAndValidate(__PEN_APP_DIR__)
if (!moduleRegistry)
  process.exit(1)

const rendererStore = createRouteRendererStore(moduleRegistry)
const { waitUntilExit } = render(<PenDevApp rendererStore={rendererStore} />, __PEN_INK_OPTIONS__)

import.meta.hot?.on(ROUTES_CHANGED_EVENT, async () => {
  try {
    const moduleRegistry = await createModuleRegistryAndValidate(__PEN_APP_DIR__)
    if (!moduleRegistry)  // keep rendering with the previous state
      return
    else  // updating the store lets React handle rerendering
      rendererStore.setState(() => createRouteRenderer(moduleRegistry))
  }
  catch (err) {
    console.error(err) // a failed reload shouldn't take the current render down with it
  }
})
await waitUntilExit()
