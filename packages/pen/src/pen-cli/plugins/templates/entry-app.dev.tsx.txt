// #region codegen
/// <reference types="vite/client" />
declare const __PEN_INK_OPTIONS__: import('ink').RenderOptions
declare const __PEN_INITIAL_PATHS__: Record<'page' | 'layout' | 'loading' | 'error' | 'default', string[]>
// #endregion

import { render } from 'ink'
import { PenDevApp, createRouteRendererStore, ROUTES_CHANGED_EVENT, createRouteRenderer, assembleModuleRegistry } from '@idlesummer/pen/internal'

// No static glob here, unlike entry-app.build.tsx.txt - its mere presence
// in a module Vite transforms is enough for Vite to track these files as
// a dependency, whether or not the result is ever used. Deleting one
// would then make Vite fully re-run this whole module on its own, which
// breaks the hot listener registered below for the rest of the session
// (new events stop reaching it afterward). Loading paths dynamically, by
// request, sidesteps that tracking entirely.
const initialPaths = __PEN_INITIAL_PATHS__
type RoutePaths = typeof initialPaths

async function loadModules(paths: string[]) {
  const entries = paths.map(async path => [path, await import(/* @vite-ignore */ `/__PEN_APP_DIR__/${path}`)] as const)
  return Object.fromEntries(await Promise.all(entries))
}

function loadRegistry(paths: RoutePaths) {
  return assembleModuleRegistry(paths, loadModules)
}
const moduleRegistry = await loadRegistry(initialPaths)
const rendererStore = createRouteRendererStore(moduleRegistry)
const { waitUntilExit } = render(<PenDevApp rendererStore={rendererStore} />, __PEN_INK_OPTIONS__)

import.meta.hot?.on(ROUTES_CHANGED_EVENT, async (paths: RoutePaths) => {
  try {
    const newModuleRegistry = await loadRegistry(paths)
    rendererStore.setState(() => createRouteRenderer(newModuleRegistry))
  }
  catch (err) {
    console.error(err)  // a failed reload shouldn't take the current render down with it
  }
})

await waitUntilExit()
