// #region codegen
/// <reference types="vite/client" />
declare const __PEN_INK_OPTIONS__: import('ink').RenderOptions
// #endregion

import type { ModuleRegistry } from '@idlesummer/pen/internal'
import { render } from 'ink'
import { PenApp, createRouteRenderer } from '@idlesummer/pen/internal'

// ─────────────────────────────────────────────────────────────────────────
// Main entry point
// ─────────────────────────────────────────────────────────────────────────

// Step 1: Load and validate the routes with Vite's builtin glob `import.meta.glob`
const moduleRegistry: ModuleRegistry = {  // `import.meta.glob` needs literal arguments
  page:     import.meta.glob('./**/page.tsx',    { base: './__PEN_APP_DIR__', eager: true }),
  layout:   import.meta.glob('./**/layout.tsx',  { base: './__PEN_APP_DIR__', eager: true }),
  loading:  import.meta.glob('./**/loading.tsx', { base: './__PEN_APP_DIR__', eager: true }),
  error:    import.meta.glob('./**/error.tsx',   { base: './__PEN_APP_DIR__', eager: true }),
  default:  import.meta.glob('./**/default.tsx', { base: './__PEN_APP_DIR__', eager: true }),
}

// Step 2: Start the app and wait for it to exit
const routeRenderer = createRouteRenderer(moduleRegistry)
const { waitUntilExit } = render(<PenApp routeRenderer={routeRenderer} />, __PEN_INK_OPTIONS__)
await waitUntilExit()
