import { useState } from 'react'
import { Box, Text, useAnimation, useApp, useInput } from 'ink'

export default function Page() {
  const { exit } = useApp()
  const { time } = useAnimation({ interval: 1000 })
  const [count, setCount] = useState(0)

  useInput((input) => {
    if (input === ' ')
      setCount(n => n + 1)
    else if (input === 'q')
      exit()
  })

  return (
    <Box flexDirection="column" borderStyle="round" paddingX={2} paddingY={1}>
      <Text bold color="cyan">Welcome to pen</Text>
      <Text dimColor>Edit this file and save, and this page updates live.</Text>

      <Box marginTop={1} flexDirection="column">
        <Text>Up for <Text bold>{Math.floor(time / 1000)}s</Text></Text>
        <Text>Count is <Text bold color="green">{count}</Text></Text>
      </Box>

      <Box marginTop={1}>
        <Text dimColor>space count · q quit</Text>
      </Box>
    </Box>
  )
}
